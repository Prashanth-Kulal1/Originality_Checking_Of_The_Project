async function loadTeamProgress() {
    const teamName = document.getElementById("progressTeamSelect").value;
    const container = document.getElementById("progressContainer");

    if (!teamName) {
        container.innerHTML = "";
        return;
    }

    const token = localStorage.getItem("token");
    const res = await fetch("/api/dashboard/coordinator", {
        headers: { "Authorization": `Bearer ${token}` }
    });

    const data = await res.json();
    const team = (data.teams || []).find(t => t.team_name === teamName);

    if (!team) {
        container.innerHTML = "<p>No data found for this team.</p>";
        return;
    }

    const progress = team.progress || [];

    if (progress.length === 0) {
        container.innerHTML = "<p>No progress uploaded yet.</p>";
        return;
    }

    let html = `
    <table>
      <thead>
        <tr>
          <th>File</th>
          <th>Notes</th>
          <th>Submitted At</th>
          <th>Download</th>
        </tr>
      </thead>
      <tbody>
    `;

    progress.forEach(item => {
        const filename = item.file_path ? item.file_path.replace(/^.*[\\\/]/, '') : null;

        html += `
        <tr>
          <td>${item.file_name || "No File"}</td>
          <td>${item.notes || "-"}</td>
          <td>${new Date(item.submitted_at).toLocaleString()}</td>
          <td>
            ${filename ? `
              <a 
                href="/api/download_file/${filename}" 
                target="_blank" 
                style="color:#2575fc;font-weight:600;">
                Download
              </a>
            ` : "No File"}
          </td>
        </tr>`;
    });

    html += "</tbody></table>";

    container.innerHTML = html;
}

    async function fetchCoordinatorData() {
      const token = localStorage.getItem("token");
      if (!token) {
        logout();
        return { teams: [], faculties: [], allocations: [] };
      }

      try {
        const res = await fetch('/api/dashboard/coordinator', {
          headers: { "Authorization": `Bearer ${token}` }
        });

        if (!res.ok) {
          if (res.status === 401 || res.status === 403) logout();
          console.error("Failed to fetch coordinator data.");
          return { teams: [], faculties: [], allocations: [] };
        }

        return await res.json();
      } catch (err) {
        console.error("Server error fetching coordinator data:", err);
        return { teams: [], faculties: [], allocations: [] };
      }
    }

    async function monitorAllocations() {
      const data = await fetchCoordinatorData();
      const tableBody = document.querySelector("#monitorTable tbody");
      tableBody.innerHTML = "";

      // Update topbar with coordinator name
      if (data.coordinator_name) {
        document.querySelector(".topbar h1").innerText = `Welcome, ${data.coordinator_name}`;
        document.querySelector(".topbar p b").innerText = data.coordinator_name;
      }

      const teamAllocationMap = {};
      const allocationData = data.allocations || [];
      allocationData.forEach(alloc => {
        teamAllocationMap[alloc.team_name] = alloc.faculty_name;
      });

      const teamsData = data.teams || [];
      if (teamsData.length === 0) {
        tableBody.innerHTML = '<tr><td colspan="5">No teams registered.</td></tr>';
        return;
      }

      teamsData.forEach(team => {
        let latestProgress = "No updates";

          if (team.progress && team.progress.length > 0) {
          // Sort by submitted_at (correct field)
          const sortedProgress = team.progress.sort(
              (a, b) => new Date(b.submitted_at) - new Date(a.submitted_at)
          );
          
          const lastProgress = sortedProgress[0];

          const noteText = lastProgress.notes
              ? lastProgress.notes.substring(0, 30)
              : "";

          const fileText = lastProgress.file_name
              ? `File: ${lastProgress.file_name}`
              : "";

          latestProgress = [noteText, fileText].filter(Boolean).join(" | ");
      }


        const proposalApproval = team.approvals
          ? team.approvals.find(a => a.stage === 'Proposal' || a.stage === 'Idea')
          : null;

        // FIXED CODE
        // FIXED Approval Status Logic
        let approvalStatus;

        if (team.project_idea && team.project_idea.status) {
            const status = team.project_idea.status.trim().toLowerCase();

            switch (status) {
                case "approved":
                    approvalStatus = `<span class="team-badge" style="background:#10b981; color:#fff">Approved</span>`;
                    break;

                case "rejected":
                    approvalStatus = `<span class="team-badge" style="background:#ef4444; color:#fff">Rejected</span>`;
                    break;

                default:
                    approvalStatus = `<span class="team-badge" style="background:#f59e0b; color:#fff">Pending</span>`;
            }
        } else {
            approvalStatus = `<span class="team-badge" style="background:#f59e0b; color:#fff">Pending</span>`;
        }




        const assignedFaculty = teamAllocationMap[team.team_name] || 'UNALLOCATED';

        const row = `
          <tr>
            <td><b>${team.team_name}</b></td>
            <td>${team.leader_name}</td>
            <td>${assignedFaculty}</td>
            <td>${latestProgress.substring(0, 30)}${latestProgress.length > 30 ? '...' : ''}</td>
            <td>${approvalStatus}</td>
          </tr>`;

        tableBody.innerHTML += row;
      });

    }

    function getApprovalColor(status) {
  if (!status) return '#f59e0b';  // Pending
  const s = status.toLowerCase();
  if (s === 'approved') return '#10b981';  // green
  if (s === 'rejected') return '#ef4444';  // red
  return '#f59e0b';  // default orange for Pending or others
}


    async function autoAllocate() {
      try {
        const token = localStorage.getItem("token");
        const response = await fetch('/api/allocate', {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'Authorization': `Bearer ${token}`
          }
        });

        const data = await response.json();
        const container = document.getElementById("facultyAllocations");
        container.innerHTML = "";

        if (!data.allocations || data.allocations.length === 0) {
          container.innerHTML = "<p>No allocations found.</p>";
          return;
        }

        const facultyGroups = {};
        data.allocations.forEach(a => {
          if (!facultyGroups[a.faculty_name]) {
            facultyGroups[a.faculty_name] = {
              email: a.faculty_email,
              teams: []
            };
          }
          facultyGroups[a.faculty_name].teams.push(a.team_name);
          facultyGroups[a.faculty_name][a.team_name] = a.match_percentage;
        });

        for (const [name, info] of Object.entries(facultyGroups)) {
          const teamsHTML = info.teams
  .map(teamName => {
      const percent = info[teamName] ?? 0;

      let color = "#64748b"; // slate
    if (percent >= 70) color = "#22c55e";      // neon green
    else if (percent >= 40) color = "#0ea5e9"; // neon blue
    else if (percent > 0) color = "#ec4899";   // neon pink

return `
  <div style="
    background: #ffffff;
    border: 2px solid ${color};
    padding: 12px 18px;
    margin: 8px 0;
    border-radius: 14px;
    display: flex;
    justify-content: space-between;
    align-items: center;
    box-shadow: 0 6px 18px rgba(0,0,0,0.08);
    font-weight: 600;
    font-size: 14px;
    transition: 0.3s ease;
  ">
    <span style="
      color:#334155;
      letter-spacing: 0.3px;
    ">
      ${teamName}
    </span>

    <span style="
      background: ${color};
      color: #ffffff;
      padding: 6px 14px;
      border-radius: 10px;
      font-size: 13px;
      min-width: 48px;
      text-align: center;
      font-weight: 700;
      box-shadow: 0 2px 6px rgba(0,0,0,0.12);
    ">
      ${percent}%
    </span>
  </div>
`;



      })
      .join('');


          const card = `
            <div class="faculty-card">
              <h3>${name}</h3>
              <p><b>Email:</b> ${info.email}</p>
              <div><b>Assigned Teams:</b><br>${teamsHTML}</div>
            </div>`;
          container.innerHTML += card;
        }

        alert("✅ " + data.message);
        monitorAllocations();
      } catch (err) {
        console.error("Error:", err);
        alert("⚠️ Error running allocation. Check backend logs.");
      }
    }

    async function loadTeamMarks() {
    const teamName = document.getElementById("teamSelect").value;
    const container = document.getElementById("marksContainer");

    if (!teamName) {
        container.innerHTML = "";
        return;
    }

    const token = localStorage.getItem("token");
    const res = await fetch("/api/dashboard/coordinator", {
        headers: { "Authorization": `Bearer ${token}` }
    });

    const data = await res.json();
    const team = (data.teams || []).find(t => t.team_name === teamName);

    if (!team || !team.marks || team.marks.length === 0) {
        container.innerHTML = "<p>No marks available for this team.</p>";
        return;
    }

    let html = "";

    team.marks.forEach(assessment => {
        const round = assessment.assessment || "Unknown Round";
        const date = new Date(assessment.date).toLocaleString();
        const memberMarks = assessment.members_marks || [];

        html += `

            <h3 style="color:#6a11cb; margin-bottom:6px; font-size:1.1em;">${round.toUpperCase()}</h3>

              <thead>
                <tr style="background:#6a11cb; color:white;">
                  <th style="padding:6px 10px; text-align:left;">Member</th>
                  <th style="padding:6px 10px; text-align:center;">Marks</th>
                </tr>
              </thead>
              <tbody>
        `;

        memberMarks.forEach(m => {
            html += `
              <tr style="border-bottom:1px solid #ddd;">
                <td style="padding:6px 10px;">${m.member}</td>
                <td style="padding:6px 10px; text-align:center;">${m.marks}</td>
              </tr>`;
        });

        html += `
              </tbody>
            </table>
          </div>
        `;
    });

    container.innerHTML = html;
}




// Call when Approval card is opened, or use the Refresh button
async function loadAllIdeas() {
  const token = localStorage.getItem("token");
  if (!token) { logout(); return; }

  const container = document.getElementById("approvalsContainer");
  container.innerHTML = "<p>Loading ideas...</p>";

  try {
    const res = await fetch("/api/ideas/all", {
      headers: { "Authorization": `Bearer ${token}` }
    });
    const payload = await res.json();
    if (!res.ok) {
      container.innerHTML = `<p style="color:red">${payload.error || 'Failed to load ideas'}</p>`;
      return;
    }

    const ideas = payload.ideas || [];
    if (ideas.length === 0) {
      container.innerHTML = "<p>No ideas submitted yet.</p>";
      return;
    }

    container.innerHTML = ""; // clear
    ideas.forEach(i => {
      const color = i.status && i.status.toLowerCase().includes("approve") ? "#10b981"
                  : i.status && i.status.toLowerCase().includes("reject") ? "#ef4444"
                  : "#f59e0b";

      const card = document.createElement("div");
      card.className = "idea-card";
      card.style.width = "520px";
      card.innerHTML = `
        <h3 style="margin-bottom:6px;">📌 ${escapeHtml(i.team_name)}</h3>
        <p style="margin:4px 0;"><b>Faculty:</b> ${escapeHtml(i.faculty_name || 'Not Assigned')}</p>
        <p style="margin:4px 0;"><b>Title:</b> ${escapeHtml(i.title || '')}</p>
        <p style="margin:6px 0;"><b>Abstract:</b> ${escapeHtml(i.abstract || '')}</p>
        <p style="margin:4px 0;"><b>Originality:</b> ${i.originality_score ?? '-'}% &nbsp; <b>Similarity:</b> ${i.similarity_percent ?? '-'}%</p>
        <p style="margin:6px 0;"><b>Status:</b> <span style="color:${color}; font-weight:700;">${escapeHtml(i.status || 'Pending')}</span></p>

        <div style="display:flex; gap:8px; margin-top:8px;">
          <button onclick="handleIdeaAction('${encodeURIComponent(i.team_name)}','${encodeURIComponent(i.title)}','Approved')" style="flex:1; background:#10b981; color:white; border:none; padding:8px; border-radius:8px; cursor:pointer;">Approve</button>
          <button onclick="handleIdeaAction('${encodeURIComponent(i.team_name)}','${encodeURIComponent(i.title)}','Rejected')" style="flex:1; background:#ef4444; color:white; border:none; padding:8px; border-radius:8px; cursor:pointer;">Reject</button>
        </div>
      `;
      container.appendChild(card);
    });

  } catch (err) {
    console.error("Error loading ideas:", err);
    container.innerHTML = "<p style='color:red'>Server error loading ideas.</p>";
  }
}

// helper to call update endpoint and optimistically update UI
async function handleIdeaAction(teamNameEnc, titleEnc, newStatus) {
  const teamName = decodeURIComponent(teamNameEnc);
  const title = decodeURIComponent(titleEnc);

  if (!confirm(`Set status of "${title}" by ${teamName} to ${newStatus}?`)) return;

  const token = localStorage.getItem("token");
  const res = await fetch("/api/ideas/update_status", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "Authorization": `Bearer ${token}`
    },
    body: JSON.stringify({ team_name: teamName, title: title, status: newStatus })
  });

  const data = await res.json();
  if (!res.ok) {
    alert(data.error || "Failed to update status.");
    return;
  }

  alert(data.message || "Status updated.");
  // refresh list
  loadAllIdeas();
}

// small helper to prevent XSS
function escapeHtml(str) {
  if (!str) return "";
  return String(str).replace(/[&<>"'`=\/]/g, function(s) {
    return ({
      '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;',
      "'": '&#39;', '/': '&#x2F;', '`': '&#x60;', '=': '&#x3D;'
    })[s];
  });
}
